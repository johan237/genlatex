\newcommand{\cvsection}[1]{%
  \vspace{10pt}

  \par\noindent\textcolor{primary}{\rule{\linewidth}{2pt}}
  \vspace{5pt}
}

\newcommand{\cvsubsection}[1]{%
  \vspace{5pt}

}

\newcommand{\cvitem}[4]{%
  \textbf{#1} \hfill \textcolor{secondary}{\textit{#2}} \\
  \textit{#3} \hfill \textcolor{accent}{#4} \\
  \vspace{2pt}
}

\newcommand{\cvskill}[2]{%
  \textbf{#1} \hfill
  \begin{tikzpicture}[baseline=-0.6ex]
    \foreach \i in {1,...,5} {
      \ifnum\i>#2
        \filldraw[lightgray] (\i*0.3,0) circle (0.1);
      \else
        \filldraw[primary] (\i*0.3,0) circle (0.1);
      \fi
    }
  \end{tikzpicture}
  \\[2pt]
}
